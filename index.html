function doPost(e) { try { // 1. Abrir la hoja de cálculo por ID o por defecto var ss = SpreadsheetApp.getActiveSpreadsheet(); var sheet = ss.getSheets()[0]; // 2. Extraer parámetros con limpieza de datos // Usamos e.parameter para asegurar compatibilidad con URLSearchParams var nombre = e.parameter.nombre || "No especificado"; var email = e.parameter.email || "No especificado"; var puntaje = Number(e.parameter.puntaje) || 0; var respuestas = e.parameter.respuestas || "{}"; var fecha = new Date(); // Columna A // 3. Lógica de Calificación var estado = ""; var colorFondo = ""; if (puntaje >= 320) { estado = "APTO - ALTA CUALIFICACIÓN"; colorFondo = "#22c55e"; // Verde } else if (puntaje >= 240) { estado = "INTERMEDIO - REQUIERE EVALUACIÓN"; colorFondo = "#eab308"; // Amarillo } else { estado = "NO APTO"; colorFondo = "#ef4444"; // Rojo } // 4. ORDEN ESTRICTO DE COLUMNAS SEGÚN TU PLANILLA: // A: Fecha y Hora // B: Nombre // C: Email // D: Detalles (Respuestas) // E: Puntaje sumado // F: Calificación var newRow = [ fecha, // Columna A nombre, // Columna B email, // Columna C respuestas, // Columna D puntaje, // Columna E estado // Columna F ]; sheet.appendRow(newRow); // 5. Aplicar formato visual a la Calificación (Columna F = 6) var lastRow = sheet.getLastRow(); var range = sheet.getRange(lastRow, 6); range.setBackground(colorFondo); range.setFontColor("#ffffff"); range.setFontWeight("bold"); range.setHorizontalAlignment("center"); // 6. Enviar Notificación por Email enviarNotificacionATitular(nombre, email, estado, puntaje, respuestas); return ContentService.createTextOutput("Success").setMimeType(ContentService.MimeType.TEXT); } catch (error) { return ContentService.createTextOutput("Error: " + error.toString()).setMimeType(ContentService.MimeType.TEXT); } } function enviarNotificacionATitular(nombre, email, estado, puntaje, respuestas) { var miEmail = "marceloholzhaus00@gmail.com"; var asunto = "🚀 NUEVA EVALUACIÓN: " + nombre + " (" + estado + ")"; var cuerpo = "Se ha registrado una nueva respuesta en Fórmula Momentum:\n\n" + "👤 Nombre: " + nombre + "\n" + "📧 Email: " + email + "\n" + "📊 Puntaje: " + puntaje + " / 400\n" + "🏆 Calificación: " + estado + "\n\n" + "📝 Detalle de Respuestas:\n" + respuestas + "\n\n" + "Los datos han sido organizados en tu planilla de Google Sheets."; MailApp.sendEmail(miEmail, asunto, cuerpo); }
